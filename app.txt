# ---------- Normalize Product Descriptions ----------
def normalize_description(text):
    """Use an LLM to summarize or standardize product info into functional terms."""
    prompt = f"""
    Extract, normalize and summarize the following software product or license string into json text of
    1. vendor - precise vendor name
    2. product_name - concise, normalized commercial product name
    3. functionalities - return concise, functionality list of the software. try to not mention vendor-specific wording or specific vendor tool, while ok to return wording of industry-standard protocols.
    
    
    Example: 
      Input: 'Figma Organization license for PointX project 2025 3 seats purchaseID AA22343'
      Output: '{{ "vendor": "Figma", 
               "product_name": "Figma Design", 
               "functionalities": "1. Design & Layout

1.1. Vector drawing and editing (shapes, paths, text, fills, strokes)
1.2. Frames, artboards, and multi-page canvases for organizing screens
1.3. Auto-layout and responsive constraints for adaptive design
1.4. Shared styles and design tokens (color, typography, spacing)
1.5. Component libraries and design systems with reusable elements and variants

2. Prototyping & Interactivity

2.1. Interactive prototypes linking screens with transitions and gestures
2.2. Animation and micro-interaction support
2.3. Asset export to multiple formats and resolutions

3. Collaboration & Developer Handoff

3.1. Real-time multi-user editing with commenting and feedback
3.2. Developer handoff tools to inspect properties, export assets, and view code specs
3.3. Version history, branching, and merging for controlled iteration

4. Platform, Extensions & Integrations

4.1. Cloud-based and desktop access
4.2. Plugin and extension ecosystem for added functionality
4.3. Integrations with third-party tools (issue tracking, version control, design ops)
4.4. API and connectors for workflow automation and data exchange

5. Enterprise & Governance
5.1. Workspace, role, and permission management
5.2. Security features such as SSO, access control, and link sharing settings
5.3. Analytics and reporting on design system usage and collaboration"    }}'

    Input: {text}
    Output:
    """
    response = client.responses.create(
        model="gpt-4.1-mini", 
        input=prompt,
        temperature=0.3
    )
    normalized_result = response.output_text.strip()

    normalized_data = {} # This will hold our final dict
        
    # 1b. Check the type of the result
    if isinstance(normalized_result, dict):
        # It's already a dictionary, great!
        normalized_data = normalized_result
    elif isinstance(normalized_result, str):
        # It's a string, so we assume it's JSON and try to parse it
        try:
            normalized_data = json.loads(normalized_result)
        except json.JSONDecodeError:
            print(f"Error: Could not parse JSON string for: '{prompt}'")
            normalized_data = {
                "vendor": "Parse Error", 
                "product_name": prompt, 
                "functionalities": "JSON Decode Error"
            }
    else:
        # It's some other unexpected type
        print(f"Error: Unexpected data type from normalize_description for: '{prompt}'")
        normalized_data = {
            "vendor": "Type Error", 
            "product_name": prompt, 
            "functionalities": "Unexpected Data Type"
        }
    print(normalized_data)
    return normalized_data 

def cut_text_to_two_lines(text: str) -> str:
    """
    Truncates a multi-line string to a maximum of two lines.

    An ellipsis (...) is appended if the original text contained more than
    two lines, indicating that content was cut.

    Args:
        text: The input string, potentially containing multiple lines.

    Returns:
        The truncated string, with a maximum of two lines and an optional ellipsis.
    """
    
    # 1. Split the text into lines
    lines = text.split('\n')
    
    # 2. Check if the text is longer than 2 lines
    if len(lines) > 4:
        # a. Take only the first two lines
        truncated_lines = lines[:4]
        
        # b. Join the lines and append the ellipsis
        return '\n'.join(truncated_lines) + '...'
    
    # 3. If the text is 2 lines or less, return it unchanged
    return text

# ---------- EMBEDDINGS ----------
@st.cache_resource
def embed_texts(texts):
    """Create embeddings for a list of texts."""
    res = client.embeddings.create(model=EMBED_MODEL, input=texts)
    return [np.array(d.embedding, dtype="float32") for d in res.data]

@st.cache_resource
def get_embeddings(df):
    """
    Create embeddings for the catalog.
    This function is cached.
    """
    print("--- Running get_embeddings() ---")
    # Combine the three columns as discussed
    texts_to_embed = (df["vendor"] + " " + df["product_name"] + " " + df["functionalities"]).tolist()
    return embed_texts(texts_to_embed)

@st.cache_resource
def build_faiss_index(embeddings):
    """
    Builds the FAISS index from a list of embeddings.
    This function is cached.
    """
    print("--- Running build_faiss_index() ---")
    dim = len(embeddings[0])
    index = faiss.IndexFlatIP(dim)
    mat = np.vstack(embeddings)
    faiss.normalize_L2(mat)
    index.add(mat)
    return index

# --- Main script logic to build index ---
# 1. Get the cached embeddings
catalog_embeddings = get_embeddings(catalog)
# 2. Get the cached index
index = build_faiss_index(catalog_embeddings)

# ---------- SEARCH ----------
def search_catalog(query, top_k=TOP_K):
    q_emb = embed_texts([query])[0].astype("float32").reshape(1, -1)
    faiss.normalize_L2(q_emb)
    D, I = index.search(q_emb, top_k)
    results_list = []
    for dist, idx in zip(D[0], I[0]):
        row = catalog.iloc[idx]
        results_list.append({
            "product_name": row.product_name,
            "vendor": row.vendor,
            "functionalities": row.functionalities,
            "input_description": row.input_description,
            "similarity": round(float(dist) * 100, 2)
        })
    results = pd.DataFrame(results_list)

    return results
